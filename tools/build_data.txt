def nfl_probs(h, a, n=8000):
    T = {t["ab"]: t for t in out["nfl"]["nfl"]["teams"]}; H, A = T[h], T[a]
    eh = max(6, 22.5 + H["x"] + A["y"] + .9); ea = max(6, 22.5 + A["x"] + H["y"] - .9)
    def drv(E):
        k = E / (11 * 1.96); td, fg = .22 * k, .14 * k; s_ = td + fg
        return (td * .85 / s_, fg * .85 / s_) if s_ > .85 else (td, fg)
    dh, da = drv(eh), drv(ea); w = 0
    for _ in range(n):
        D = 10 + _r.randint(0, 2); ph = pa = 0
        for _ in range(D):
            r = _r.random(); ph += 7 if r < dh[0] else 3 if r < dh[0] + dh[1] else 0
            r = _r.random(); pa += 7 if r < da[0] else 3 if r < da[0] + da[1] else 0
        w += 1 if ph > pa else .5 if ph == pa else 0
    return w / n, 0.0, 1 - w / n
